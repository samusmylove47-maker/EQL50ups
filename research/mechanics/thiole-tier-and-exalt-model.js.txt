/* ============================================================
   ITEM TIERS 0-10
   Source: eqlwiki.com/Item_Upgrade_System.

   These are NOT scraped, and there is nothing to scrape — every item page on
   the wiki carries one stat block, at tier 0. Tiers are *computed*:

     - each tier is a cumulative +10% to the item's stats
     - the bonus tracks item XP, not the tier number, so it creeps up mid-tier
     - everything rounds down
     - a minimum +1 per stat is guaranteed at the *start of each tier* only;
       mid-tier creep does not get it
     - weight shrinks the same way, floor 0.1
     - a minimum +1 per tier is guaranteed on STATS ONLY, and it is cumulative
     - weapon damage takes the same +10% but gets no minimum; delay never moves

   Verified against the wiki's own worked example (Umbral Platemail Breastplate,
   AC 30 -> 33 at T1 -> 34 at T1+1xp -> 36 at T2).
   ============================================================ */
const TIER_XP   = [0,1,3,7,15,31,63,127,255,511,1023];   // total xp to sit at tier N
const TIER_NEXT = [1,2,4,8,16,32,64,128,256,512,null];   // xp from tier N to N+1
const MAX_XP = 1023;

/* What one merge is worth. Gear gives the xp it took to reach its own tier;
   motes give a fixed amount but refuse to feed an item above their cap. */
const MOTES = [
  {n:"Infinitesimal",max:0,xp:1},  {n:"Minor",max:1,xp:1},   {n:"Lesser",max:2,xp:2},
  {n:"Potential",max:3,xp:4},      {n:"Major",max:4,xp:5},   {n:"Greater",max:5,xp:6},
  {n:"Superior",max:6,xp:7},       {n:"Grand",max:7,xp:8},   {n:"Ascendant",max:8,xp:9},
  {n:"Infinite",max:9,xp:10},
];

const tierOf = xp => { let t = 0; for(let i=0;i<11;i++) if(xp >= TIER_XP[i]) t = i; return t; };

/** cumulative stat bonus as a fraction, e.g. 0.15 for +15% */
function cumBonus(xp){
  const t = tierOf(xp);
  if(t >= 10) return 1;
  const into = xp - TIER_XP[t], need = TIER_NEXT[t];
  return (t*10 + (into/need*100)/10) / 100;
}

/** one stat, tier-0 value -> value at this xp.
 *
 * The +1-per-tier guarantee IS cumulative here, and it applies to stats only.
 * That's what makes a +5 STR item read 9 at +4 rather than 7: the percentage
 * contributes +2 over four tiers, the guarantee contributes +4, and the
 * larger wins. On big values it never binds — AC 30 gains +3 a tier from the
 * percentage alone, which is why the wiki's Umbral Platemail example is
 * silent about it.
 *
 * Weapon damage does NOT get this. See scaleDmg — that's the whole difference
 * between the two, and it's why a 5-damage weapon reads 7 at +4 while a +5
 * stat on the same item reads 9.
 *
 * Verified against the wiki's published example: Umbral Platemail Breastplate
 * AC 30 -> 33 at T1, 34 at T1+1xp, 36 at T2.
 */
function scaleStat(v, xp){
  if(!v) return v;
  const t = tierOf(xp), cum = cumBonus(xp);
  if(v > 0){
    const atTier = Math.max(Math.floor(v * (1 + t/10)), v + t);  // +1/tier guarantee
    return Math.max(atTier, Math.floor(v * (1 + cum)));          // mid-tier creep
  }
  /* UNVERIFIED: the wiki publishes no negative-stat example. It says the +1 is a
     bonus and that everything rounds down; applied literally that deepens a
     penalty as you upgrade. That is what this does. If you see otherwise
     in-game, this is the line to change. */
  return -Math.floor(Math.abs(v) * (1 + cum));
}

/* Haste is the one channel where +1 a tier IS cumulative, and it's observed,
   not inferred: the wiki's slider on Cloak of Flames ladders 36, 37, 38 ...
   46 across the ten tiers. It takes no percentage at all — 10% would put it
   at 72 and 5% at 54, and neither is what the slider shows. */
const NO_PCT_STATS = new Set(["HASTE"]);
function scaleHaste(v, xp){ return v ? v + tierOf(xp) : v; }

/* Weapon damage takes the same +10% a tier as stats, but gets NO minimum
   guarantee — the wiki's "+1 at the start of each tier" is a stats rule and
   damage is not covered by it. So this is the percentage alone, rounded down,
   with mid-tier creep and nothing else.

   This is the entire difference between damage and stats. A 5-damage weapon
   reads 7 at +4 (percentage only: floor(5 * 1.4)), while a +5 stat on the
   same item reads 9 (guarantee: 5 + 4). A tier can legitimately do nothing
   to weapon damage — a 2-damage weapon sits at 2 through five tiers.

   The rate is pinned by that same observation: at 5% a 5-damage weapon would
   read 6 at +4, so eqlwiki's "+5% per Tier" bullet for damage stays
   rejected. */
function scaleDmg(v, xp){
  if(!v) return v;
  if(v < 0) return -Math.floor(Math.abs(v) * (1 + cumBonus(xp)));
  return Math.max(Math.floor(v * (1 + tierOf(xp)/10)),
                  Math.floor(v * (1 + cumBonus(xp))));
}

function scaledItem(item, xp){
  const stats = {};
  for(const [k,v] of Object.entries(item.stats||{}))
    stats[k] = NO_PCT_STATS.has(k) ? scaleHaste(v, xp) : scaleStat(v, xp);
  const o = {stats, tier:tierOf(xp), xp, cum:cumBonus(xp)};
  if(item.dmg) o.dmg = scaleDmg(item.dmg, xp);
  if(item.delay) o.delay = item.delay;                 // never scales
  if(item.wt != null) o.wt = Math.max(0.1, Math.round(item.wt * (1 - cumBonus(xp)) * 10) / 10);
  return o;
}
